<?php

/**
 * Server monitoring agent (v2 - Heartbeat protocol)
 *
 * Protocol:
 * 1. Read bootstrap.json (written by installer)
 * 2. Register via POST /api/v1/register using the provision token
 * 3. Loop: collect metrics, send heartbeat via POST /api/v1/agent/heartbeat
 *    with Bearer token auth
 * 4. Handle pending commands from heartbeat response
 */

$bootstrapPath = __DIR__ . '/bootstrap.json';
$config = json_decode(file_get_contents($bootstrapPath), true);

if (!$config) {
    fwrite(STDERR, "Failed to read bootstrap.json\n");
    exit(1);
}

// -----------------------------------------------------------------------
// Networking helpers
// -----------------------------------------------------------------------

function hasInternet(): bool
{
    $ch = curl_init('https://1.1.1.1');
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_NOBODY => true,
        CURLOPT_TIMEOUT => 5,
        CURLOPT_CONNECTTIMEOUT => 5,
    ]);
    curl_exec($ch);
    $ok = curl_errno($ch) === 0;
    curl_close($ch);
    return $ok;
}

function sendWithRetry(string $url, array $payload, array $extraHeaders = [], int $maxAttempts = 0)
{
    $attempt = 0;
    $delay = 2;
    $maxDelay = 60;

    while (true) {
        $attempt++;

        $ch = curl_init();
        curl_setopt_array($ch, [
            CURLOPT_URL => $url,
            CURLOPT_POST => true,
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_TIMEOUT => 15,
            CURLOPT_CONNECTTIMEOUT => 10,
            CURLOPT_HTTPHEADER => array_merge(
                ['Content-Type: application/json'],
                $extraHeaders
            ),
            CURLOPT_POSTFIELDS => json_encode($payload),
        ]);

        $response = curl_exec($ch);
        $curlError = curl_errno($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);

        if ($curlError === 0 && $httpCode >= 200 && $httpCode < 300) {
            return json_decode($response, true);
        }

        $networkIssue = $curlError !== 0;

        if ($networkIssue) {
            fwrite(STDERR, "[attempt $attempt] curl error: " . curl_strerror($curlError) . " — waiting for internet...\n");

            while (!hasInternet()) {
                sleep(5);
            }
            continue;
        }

        fwrite(STDERR, "[attempt $attempt] HTTP $httpCode from $url\n");
        $body = json_decode($response ?? '', true);
        if (isset($body['message'])) {
            fwrite(STDERR, "  Response: {$body['message']}\n");
        }

        if ($maxAttempts > 0 && $attempt >= $maxAttempts) {
            return false;
        }

        sleep($delay);
        $delay = min($delay * 2, $maxDelay);
    }
}

// -----------------------------------------------------------------------
// Hardware / spec collection
// -----------------------------------------------------------------------

function cpuSpec(): array
{
    $cpuInfo = file_get_contents('/proc/cpuinfo');
    preg_match('/model name\s*:\s*(.*)$/m', $cpuInfo, $match);
    $cpuName = trim($match[1] ?? 'Unknown CPU');
    $cores = (int) trim(shell_exec('nproc') ?? '0');

    return [
        'model' => $cpuName,
        'cores' => $cores,
    ];
}

function memorySpec()
{
    $memInfo = file_get_contents('/proc/meminfo');
    preg_match('/MemTotal:\s+(\d+)/', $memInfo, $matches);
    $totalRamGB = round((($matches[1] ?? 0) / 1024 / 1024), 2);
    return $totalRamGB . ' GB';
}

function diskSpec()
{
    $totalGB = round(disk_total_space('/') / (1024 ** 3), 2);
    return $totalGB . ' GB';
}

function osSpec()
{
    $distro = 'Unknown Linux';

    if (is_readable('/etc/os-release')) {
        $lines = file('/etc/os-release', FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES);
        foreach ($lines as $line) {
            if (str_starts_with($line, 'PRETTY_NAME=')) {
                $distro = str_replace(['PRETTY_NAME=', '"'], '', $line);
                break;
            }
        }
    } elseif (shell_exec('command -v lsb_release')) {
        $distro = trim(shell_exec('lsb_release -ds') ?? '');
    }

    return trim($distro);
}

// -----------------------------------------------------------------------
// Open ports — database services only
// -----------------------------------------------------------------------

const DB_PROCESS_NAMES = [
    'mysqld',
    'mariadbd',
    'mariadb',
    'postgres',
    'postmaster',
    'mongod',
    'redis-server',
    'memcached',
    'cassandra',
    'rabbitmq-server',
    'beam.smp',
    'influxd',
    'clickhouse-server',
    'elasticsearch',
];

const DB_DEFAULT_PORTS = [
    3306 => 'mysql/mariadb',
    5432 => 'postgresql',
    27017 => 'mongodb',
    6379 => 'redis',
    11211 => 'memcached',
    9042 => 'cassandra',
    5672 => 'rabbitmq',
    8086 => 'influxdb',
    9000 => 'clickhouse',
    9200 => 'elasticsearch',
];

function openDatabasePorts(): array
{
    $output = shell_exec('ss -tlnp 2>/dev/null');
    if (!$output) {
        $output = shell_exec('netstat -tlnp 2>/dev/null');
    }
    if (!$output) {
        return [];
    }

    $results = [];

    foreach (explode("\n", trim($output)) as $line) {
        if (stripos($line, 'State') !== false || trim($line) === '') {
            continue;
        }

        if (!preg_match('/(\S+):(\d+)\s+\S+\s+(.*)$/', $line, $m)) {
            continue;
        }

        $address = $m[1];
        $port = (int) $m[2];
        $tail = $m[3];

        $process = null;
        if (preg_match('/\(\("([^"]+)"/', $tail, $pm)) {
            $process = $pm[1];
        } elseif (preg_match('/(\d+)\/([^\s]+)/', $tail, $pm)) {
            $process = $pm[2];
        }

        $isDbProcess = $process && in_array($process, DB_PROCESS_NAMES, true);
        $isDbPort = isset(DB_DEFAULT_PORTS[$port]);

        if (!$isDbProcess && !$isDbPort) {
            continue;
        }

        $results[] = [
            'address' => $address === '0.0.0.0' ? '0.0.0.0' : $address,
            'port' => $port,
            'protocol' => 'tcp',
            'process' => $process ?? (DB_DEFAULT_PORTS[$port] ?? 'unknown'),
            'state' => 'listening',
        ];
    }

    return $results;
}

// -----------------------------------------------------------------------
// Services (systemd units)
// -----------------------------------------------------------------------

function listServices(): array
{
    $output = shell_exec('systemctl list-units --type=service --no-pager --no-legend 2>/dev/null');
    if (!$output) {
        return [];
    }

    $services = [];
    foreach (explode("\n", trim($output)) as $line) {
        if (!preg_match('/^(\S+)\s+\S+\s+\S+\s+(\S+)/', $line, $m)) {
            continue;
        }
        $services[] = [
            'identifier' => $m[1],
            'name' => $m[1],
            'state' => $m[2],
        ];
    }

    return $services;
}

// -----------------------------------------------------------------------
// Live stats (per-tick monitoring payload)
// -----------------------------------------------------------------------

function cpuUsage(): array
{
    $load = sys_getloadavg();
    return [
        'load1' => $load[0],
        'load5' => $load[1],
        'load15' => $load[2],
    ];
}

function memoryUsage(): array
{
    $data = file('/proc/meminfo');
    $mem = [];
    foreach ($data as $line) {
        [$key, $value] = explode(':', $line, 2);
        $mem[$key] = (int) filter_var($value, FILTER_SANITIZE_NUMBER_INT);
    }
    $total = $mem['MemTotal'] ?? 0;
    $available = $mem['MemAvailable'] ?? 0;
    $used = $total - $available;

    return [
        'total_kb' => $total,
        'used_kb' => $used,
        'free_kb' => $available,
        'percent' => $total > 0 ? round(($used / $total) * 100, 2) : 0,
    ];
}

function diskUsage(): array
{
    $total = disk_total_space('/');
    $free = disk_free_space('/');

    return [
        'total' => $total,
        'used' => $total - $free,
        'free' => $free,
        'percent' => $total > 0 ? round((($total - $free) / $total) * 100, 2) : 0,
    ];
}

function uptime(): float
{
    $uptime = trim(file_get_contents('/proc/uptime'));
    return (float) explode(' ', $uptime)[0];
}

function networkStats(): array
{
    $data = file_get_contents('/proc/net/dev');
    preg_match_all('/^\s*(eth\w*|en\w*):\s*(\d+)(?:\s+\d+){6}\s+(\d+)/m', $data, $matches, PREG_SET_ORDER);

    return array_map(fn($m) => [
        'interface' => $m[1],
        'rx_bytes'  => (int)$m[2],
        'tx_bytes'  => (int)$m[3],
    ], $matches);
}

function topProcesses(): array
{
    $output = shell_exec('ps -eo pid,comm,%cpu,%mem --sort=-%cpu | head -6');
    $lines = explode("\n", trim($output));
    $header = array_shift($lines);
    $result = [];
    foreach ($lines as $line) {
        $line = trim($line);
        if ($line === '') continue;
        $parts = preg_split('/\s+/', $line);
        if (count($parts) >= 4 && is_numeric($parts[0])) {
            $result[] = [
                'pid' => (int) $parts[0],
                'name' => $parts[1],
                'cpu' => (double) $parts[2],
                'memory' => (double) $parts[3],
            ];
        }
    }
    return $result;
}

// -----------------------------------------------------------------------
// Build heartbeat payload
// -----------------------------------------------------------------------

function buildHeartbeatPayload(array $config, string $identityToken): array
{
    return [
        'agent_version' => $config['agent_version'] ?? '2.0',
        'configuration_version' => $config['_configuration_version'] ?? 1,
        'timestamp' => time(),
        'hostname' => gethostname(),
        'cpu' => cpuUsage(),
        'memory' => memoryUsage(),
        'disk' => diskUsage(),
        'uptime' => uptime(),
        'network' => networkStats(),
        'top_processes' => topProcesses(),
        'open_db_ports' => openDatabasePorts(),
        'services' => listServices(),
    ];
}

// -----------------------------------------------------------------------
// Main
// -----------------------------------------------------------------------

echo "Agent v2 starting...\n";
echo "Registering with server...\n";

$registerPayload = [
    'token' => $config['token'],
    'agent_version' => $config['agent_version'] ?? '2.0',
    'hostname' => $config['hostname'] ?? gethostname(),
    'operating_system' => osSpec(),
    'architecture' => php_uname('m'),
    'cpu' => cpuSpec(),
    'memory' => memorySpec(),
    'disk' => diskSpec(),
    'capabilities' => ['metrics.cpu', 'metrics.memory', 'metrics.disk', 'metrics.network', 'metrics.processes', 'ports.scan'],
];

$registerResult = sendWithRetry($config['register_url'], $registerPayload, [], 5);

if (!$registerResult || !isset($registerResult['identity'])) {
    fwrite(STDERR, "Registration failed. Check your provision token.\n");
    exit(1);
}

$identityToken = $registerResult['identity'];
echo "Registered successfully. Identity token obtained.\n";

$heartbeatInterval = $registerResult['heartbeat_interval'] ?? ($config['heartbeat_interval'] ?? 5);

if (isset($registerResult['configuration'])) {
    $config['_configuration_version'] = $registerResult['configuration']['version'] ?? 1;
}

echo "Starting heartbeat loop (interval: {$heartbeatInterval}s)...\n";

while (true) {
    $payload = buildHeartbeatPayload($config, $identityToken);

    $response = sendWithRetry(
        $config['api_url'],
        $payload,
        ['Authorization: Bearer ' . $identityToken]
    );

    if ($response) {
        if (isset($response['heartbeat_interval'])) {
            $heartbeatInterval = (int) $response['heartbeat_interval'];
        }

        if (isset($response['configuration'])) {
            $config['_configuration_version'] = $response['configuration']['version']
                ?? $config['_configuration_version'];
            echo "Configuration updated to version {$config['_configuration_version']}\n";
        }

        if (!empty($response['pending_commands'])) {
            $completedCommands = [];
            foreach ($response['pending_commands'] as $cmd) {
                echo "Executing command: {$cmd['type']} (id: {$cmd['id']})\n";
                $result = executeCommand($cmd);
                $completedCommands[] = $result;
            }
            if (!empty($completedCommands)) {
                $payload = array_merge(buildHeartbeatPayload($config, $identityToken), [
                    'completed_commands' => $completedCommands,
                ]);
                sendWithRetry(
                    $config['api_url'],
                    $payload,
                    ['Authorization: Bearer ' . $identityToken]
                );
            }
        }
    }

    sleep($heartbeatInterval);
}

function executeCommand(array $command): array
{
    $result = [
        'command_id' => $command['id'],
        'status' => 'completed',
        'output' => '',
        'error' => null,
        'execution_time_ms' => 0,
    ];

    $start = microtime(true);

    if ($command['type'] === 'shell') {
        $cmdStr = $command['payload']['command'] ?? '';
        if ($cmdStr) {
            $output = shell_exec($cmdStr . ' 2>&1');
            $result['output'] = trim($output ?? '');
        }
    } elseif ($command['type'] === 'update_config') {
        $result['output'] = 'Configuration update acknowledged';
    }

    $result['execution_time_ms'] = (int) ((microtime(true) - $start) * 1000);

    return $result;
}
